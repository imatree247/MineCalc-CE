inline int airay_trace_point(const point& pl, const int pixelx,const int pixely,const int& anglexz,const int& angleyz,int& d) {
    flint currX = pl.x;
    flint currY = pl.y;
    flint currZ = pl.z;
    
	int pix = (int)currX;
	int piy = (int)currY;
	int piz = (int)currZ;
	
	int ix = (int)currX;
	int iy = (int)currY;
	int iz = (int)currZ;
    // Get directions from cache
    flint dx = cached_x_dir[pixelx];
    flint dy = cached_y_dir[pixely];
    flint dz = step_divide; // Looking forward
    // Step the ray 10 times in small increments            16
	rot_point_3d(dx,dy,dz,0,0,0,anglexz,angleyz);
	//printf("%f\n",(float)myst);
    for (int i = 0; i < 30; i++) {
        // Increment by the direction vector
        currX += dx; 
        currY -= dy;
        currZ += dz;
		pix = ix;
		piy = iy;
		piz = iz;		
		ix = (int)currX;
		iy = (int)currY;
		iz = (int)currZ;

        // Check bounds
        if (ix < 0 || ix >= 6 || iy < 0 || iy >= 6 || iz < 0 || iz >= 6) 
            return 0;

        if (vis_map(ix,iy,iz) != 0){
			d=piy!=iy? 1:d;
			d=pix!=ix? 0:d;
			d=piz!=iz? 2:d;
			//d=0;
            return vis_map(ix,iy,iz);
		}
    }
    return 0;
}
